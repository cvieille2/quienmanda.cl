## Custom Post Type: ficha_motel (NUEVO — reemplaza ficha_destacada)
```php
<?php
// functions.php
function registrar_ficha_motel() {
    register_post_type('ficha_motel', array(
        'labels' => array('name' => 'Fichas de Motel', 'singular_name' => 'Ficha de Motel', 'add_new' => 'Nueva ficha'),
        'public' => true,
        'menu_icon' => 'dashicons-building',
        'supports' => array('title', 'editor', 'thumbnail', 'custom-fields', 'excerpt'),
        'has_archive' => true,
        'rewrite' => array('slug' => '%ciudad%', 'with_front' => false),
        'show_in_rest' => true,
    ));
}
add_action('init', 'registrar_ficha_motel');

function ficha_motel_post_link($post_link, $post) {
    if ($post->post_type === 'ficha_motel') {
        $ciudad = get_post_meta($post->ID, 'ciudad', true);
        if ($ciudad) return home_url(sanitize_title($ciudad) . '/' . $post->post_name . '/');
    }
    return $post_link;
}
add_filter('post_type_link', 'ficha_motel_post_link', 10, 2);

function shortcode_fichas_motel($atts) {
    $args = array(
        'post_type' => 'ficha_motel',
        'posts_per_page' => isset($atts['limite']) ? intval($atts['limite']) : 10,
        'meta_key' => 'precio_desde',
        'orderby' => 'meta_value_num', 'order' => 'ASC'
    );
    if (!empty($atts['ciudad'])) {
        $args['meta_query'] = array(array('key' => 'ciudad', 'value' => $atts['ciudad'], 'compare' => '='));
    }
    $fichas = new WP_Query($args);
    $html = '<div class="grid-fichas" style="display:grid;grid-template-columns:repeat(auto-fill,minmax(300px,1fr));gap:20px;padding:20px">';
    while ($fichas->have_posts()) { $fichas->the_post();
        $precio = get_post_meta(get_the_ID(), 'precio_desde', true);
        $whatsapp = get_post_meta(get_the_ID(), 'whatsapp', true);
        $servicios = get_post_meta(get_the_ID(), 'servicios', true);
        $badges = $servicios ? array_slice($servicios, 0, 3) : array();
        $html .= '<article style="border:1px solid #e0e0e0;border-radius:12px;overflow:hidden;background:#fff">';
        $html .= get_the_post_thumbnail(get_the_ID(), 'medium', array('style'=>'width:100%;height:180px;object-fit:cover'));
        $html .= '<div style="padding:16px">';
        $html .= '<h3 style="margin:0 0 8px;font-size:18px"><a href="'.get_permalink().'" style="color:#1a73e8">'.get_the_title().'</a></h3>';
        $html .= '<div style="color:#666;font-size:14px">'.get_post_meta(get_the_ID(), 'direccion', true).'</div>';
        $html .= '<div style="font-size:22px;color:#1a73e8;font-weight:bold;margin:8px 0">Desde $'.number_format(intval($precio)).'</div>';
        if ($badges) {
            foreach ($badges as $b) { $html .= '<span style="background:#e8f0fe;color:#1a73e8;padding:2px 8px;border-radius:12px;font-size:12px;margin-right:4px">'.$b.'</span>'; }
        }
        if ($whatsapp) { $html .= '<a href="https://wa.me/'.$whatsapp.'" style="display:block;background:#25D366;color:white;text-align:center;padding:10px;border-radius:6px;text-decoration:none;margin-top:10px;font-weight:bold">WhatsApp</a>'; }
        $html .= '</div></article>';
    }
    $html .= '</div>'; wp_reset_postdata();
    return $html;
}
add_shortcode('fichas_motel', 'shortcode_fichas_motel');
?>
```

## Campos personalizados obligatorios
| Campo | Tipo | Requerido |

| nombre_motel | text | si |

| comuna | text | si |

| telefono | text | si |
| whatsapp | text | no |
| url_oficial | url | no |
| geo_lat | number | si |
| geo_lng | number | si |
| precio_desde | number | si |
| horario | text | si |
| servicios | checkbox group | si (min 5) |
| tarifas | repeater (hab, tiempo, precio, incluye) | si |
| fecha_verificacion | date | si |
| fuente_dato | select | si |
| imagenes_galeria | gallery | si |
