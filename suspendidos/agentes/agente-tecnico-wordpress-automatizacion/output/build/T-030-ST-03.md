## Custom Post Type: ficha_destacada
`php
<?php
// functions.php - Registrar CPT
function registrar_ficha_destacada() {
    register_post_type('ficha_destacada', array(
        'labels' => array('name' => 'Fichas Destacadas', 'singular_name' => 'Ficha Destacada'),
        'public' => true,
        'menu_icon' => 'dashicons-star-filled',
        'supports' => array('title', 'editor', 'thumbnail', 'custom-fields'),
        'has_archive' => true,
    ));
}
add_action('init', 'registrar_ficha_destacada');

// Shortcode [fichas_destacadas categoria="lujo" limite="6"]
function shortcode_fichas_destacadas() {
     = array(
        'post_type' => 'ficha_destacada',
        'posts_per_page' => isset(['limite']) ? intval(['limite']) : 10,
        'meta_key' => 'precio',
        'orderby' => 'meta_value_num',
        'order' => 'DESC'
    );
    if (!empty(['categoria'])) {
        ['tax_query'] = array(array('taxonomy' => 'categoria_ficha', 'field' => 'slug', 'terms' => ['categoria']));
    }
     = new WP_Query();
     = '<div style="display:grid;grid-template-columns:repeat(auto-fill,minmax(300px,1fr));gap:20px;padding:20px">';
    while (->have_posts()) { ->the_post();
         = get_post_meta(get_the_ID(), 'precio', true);
         = get_post_meta(get_the_ID(), 'whatsapp', true);
         .= '<div style="border:1px solid #ddd;border-radius:8px;padding:15px">';
         .= get_the_post_thumbnail(get_the_ID(), 'medium', array('style'=>'width:100%;height:200px;object-fit:cover;border-radius:4px'));
         .= '<h3 style="margin:10px 0">'.get_the_title().'</h3>';
         .= '<p>'.get_the_excerpt().'</p>';
         .= '<div style="font-size:24px;color:#1a73e8;font-weight:bold">$'.number_format(intval()).'/mes</div>';
        if () {  .= '<a href="https://wa.me/'..'" style="display:block;background:#25D366;color:white;text-align:center;padding:10px;border-radius:6px;text-decoration:none;margin-top:10px">Contactar por WhatsApp</a>'; }
         .= '</div>';
    }
     .= '</div>';
    wp_reset_postdata();
    return ;
}
add_shortcode('fichas_destacadas', 'shortcode_fichas_destacadas');
?>
`
